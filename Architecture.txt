>>> File: app.js
app.controller('ShoppingController', function($scope, $http) {

*** $http	- Sends HTTP requests to Java Spring Boot

    // Checkout
    $scope.checkout = function () {

        alert(
            'Order Total: RM ' +
            $scope.getTotalPrice().toFixed(2)
        );

        var orderData = {
            items: $scope.cart,
            total: $scope.total
        };

        $http.post(API_URL, orderData, {
          responseType: "text",
          transformResponse: function (data) {
          return data;
          }
        })
        .then(function (response) {

            alert(response.data);
           //example: 
           // Order processed successfully: 
           // {"items":[{"id":4,"name":"Monitor","price":650,"quantity":1}]}
        })
        .catch(function (error) {

            console.error(error);

            alert("Unable to submit order.");
        });
    };

});

---------------- CRUD Image ----------------

CREATE TABLE image_file (
    id BIGSERIAL PRIMARY KEY,
    file_name VARCHAR(255) NOT NULL,
    content_type VARCHAR(100) NOT NULL,
    image_data BYTEA NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Important version note: 
use Camel 4.18.x, not 4.19 or later, because Camel 4.19 moved to Spring Boot 4 and no longer supports Spring Boot 3. 
Camel 4.18.5 supports Java 17 and is the latest 4.18 LTS release listed by Apache Camel

How Camel works here:
- direct:image-create receives image information from the controller.
- sql: executes the SQL statement.
- :#fileName reads the fileName value from the message body's Map.
- :#imageData reads the image's byte[].
- dataSource=#dataSource uses Spring Boot's configured PostgreSQL DataSource.
